# ==============================================================================
# requirements.txt
# ------------------------------------------------------------------------------
# This file lists every third-party Python package the Insurance Virtual
# Assistant needs to run. Install them all with:
#
#       pip install -r requirements.txt
#
# Why pin versions?
#   RAG apps chain together many libraries (LangChain, ChromaDB, OpenAI SDK,
#   Streamlit, PyMuPDF...). These libraries evolve quickly and sometimes make
#   breaking changes between versions. Pinning versions means the app you build
#   today will still work the same way next month.
# ==============================================================================

# --- Frontend --------------------------------------------------------------
streamlit==1.38.0            # Web UI framework used to build the chat interface

# --- LLM + Embeddings (OpenAI) ----------------------------------------------
openai==1.51.0                # Official OpenAI Python SDK (GPT-4o + embeddings)

# --- Orchestration Framework -------------------------------------------------
langchain==0.3.1              # Core LangChain framework (chains, prompts, memory)
langchain-core==0.3.6          # Pinned to avoid incompatible resolution with newer versions
langchain-openai==0.2.1        # LangChain <-> OpenAI integration (chat + embeddings)
langchain-community==0.3.1     # Community integrations (used for some loaders/utilities)
langchain-text-splitters==0.3.0 # Text splitting utilities (RecursiveCharacterTextSplitter)
langchain-chroma==0.1.4        # LangChain <-> ChromaDB integration

# --- Vector Database ---------------------------------------------------------
chromadb==0.5.3                # Local, on-disk vector database for storing embeddings

# --- PDF Parsing ---------------------------------------------------------------
PyMuPDF==1.24.10               # Fast PDF text/metadata extraction (imported as "fitz")

# --- Local Embeddings (free, no API key needed) --------------------------------
sentence-transformers>=2.2.0      # HuggingFace local embeddings (all-MiniLM-L6-v2)

# --- HTTP client (compatibility pin) -------------------------------------------
httpx<0.28                       # openai<=1.55 passes 'proxies' to httpx.Client(); removed in 0.28

# --- Databricks Foundation Model APIs ------------------------------------------
databricks-sdk>=0.30.0           # Databricks SDK for auth + model serving calls
mlflow>=2.16.0                   # Required by langchain ChatDatabricks integration
langchain-databricks>=0.1.0      # ChatDatabricks for Databricks Foundation Model APIs

# --- Utilities -----------------------------------------------------------------
python-dotenv==1.0.1            # Loads environment variables (e.g., API keys) from .env
tiktoken==0.7.0                 # Token counting used for chunking & cost estimation
pydantic==2.9.2                 # Data validation, used internally by LangChain
tenacity==8.5.0                 # Retry logic for flaky network/API calls
